cmake_minimum_required(VERSION 3.24)
project(MusicSorter VERSION 1.4.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Multimedia Concurrent Test Network)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
add_library(music_core src/Core.cpp src/Core.h)
target_link_libraries(music_core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent)
target_include_directories(music_core PUBLIC src)
include(cmake/FFmpeg.cmake)
add_library(music_audio src/WaveformGenerator.cpp src/WaveformGenerator.h src/AudioController.cpp src/AudioController.h src/CompressedAudio.cpp src/CompressedAudio.h src/AudioRepository.cpp src/AudioRepository.h src/AudioEngine.cpp src/AudioEngine.h)
target_include_directories(music_audio SYSTEM PRIVATE third_party/ffmpeg/include)
target_link_libraries(music_audio PUBLIC music_core Qt6::Multimedia)
target_link_libraries(music_audio PRIVATE FFmpeg::avcodec FFmpeg::avformat FFmpeg::avutil FFmpeg::swresample)
add_library(music_ui src/MainWindow.cpp src/MainWindow.h src/DirectoryTree.h src/TrackGridView.cpp src/TrackGridView.h src/SettingsDialog.cpp src/SettingsDialog.h src/WindowsIntegration.cpp src/WindowsIntegration.h src/SingleInstance.cpp src/SingleInstance.h src/VisualizerController.cpp src/VisualizerController.h)
target_link_libraries(music_ui PUBLIC music_audio Qt6::Widgets Qt6::Network PRIVATE shell32)
qt_add_executable(MusicSorter WIN32 src/main.cpp resources/app.rc)
qt_add_resources(MusicSorter app_icons PREFIX "/" FILES resources/musicsorter.ico)
target_link_libraries(MusicSorter PRIVATE music_ui)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(MusicSorterPlaybackDiagnostics tests/PlaybackDiagnostics.cpp)
  set_target_properties(MusicSorterPlaybackDiagnostics PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(MusicSorterPlaybackDiagnostics PRIVATE music_ui Qt6::Test)
  target_compile_definitions(MusicSorterPlaybackDiagnostics PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME PlaybackResilience COMMAND MusicSorterPlaybackDiagnostics -o "${CMAKE_BINARY_DIR}/playback-diagnostics.txt,txt")
  set_tests_properties(PlaybackResilience PROPERTIES TIMEOUT 60)
  qt_add_executable(MusicSorterVisualizerTests tests/VisualizerTests.cpp)
  set_target_properties(MusicSorterVisualizerTests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(MusicSorterVisualizerTests PRIVATE music_ui Qt6::Test)
  target_compile_definitions(MusicSorterVisualizerTests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" VISUALIZER_EXE="${CMAKE_CURRENT_SOURCE_DIR}/integration/spectral-grid/build/Release/SpectralGrid.exe")
  add_test(NAME VisualizerTests COMMAND MusicSorterVisualizerTests -o "${CMAKE_BINARY_DIR}/visualizer-results.txt,txt" -o "-,txt")
  set_tests_properties(VisualizerTests PROPERTIES TIMEOUT 90)
  qt_add_executable(MusicSorterInstanceProbe tests/InstanceProbe.cpp src/SingleInstance.cpp src/SingleInstance.h)
  set_target_properties(MusicSorterInstanceProbe PROPERTIES WIN32_EXECUTABLE FALSE)
  target_include_directories(MusicSorterInstanceProbe PRIVATE src)
  target_link_libraries(MusicSorterInstanceProbe PRIVATE Qt6::Core Qt6::Network)
  qt_add_executable(MusicSorterTests tests/CoreTests.cpp)
  set_target_properties(MusicSorterTests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(MusicSorterTests PRIVATE music_audio Qt6::Test)
  target_compile_definitions(MusicSorterTests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME CoreTests COMMAND MusicSorterTests -o "${CMAKE_BINARY_DIR}/core-results.txt,txt" -o "-,txt")
  set_tests_properties(CoreTests PROPERTIES TIMEOUT 90)
  qt_add_executable(MusicSorterUiTests tests/UiTests.cpp)
  add_dependencies(MusicSorterUiTests MusicSorterInstanceProbe)
  set_target_properties(MusicSorterUiTests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(MusicSorterUiTests PRIVATE music_ui Qt6::Test)
  add_test(NAME UiTests COMMAND MusicSorterUiTests -o "${CMAKE_BINARY_DIR}/ui-results.txt,txt" -o "-,txt")
  set_tests_properties(UiTests PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(MusicSorterAudioTests tests/AudioTests.cpp)
  set_target_properties(MusicSorterAudioTests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(MusicSorterAudioTests PRIVATE music_audio Qt6::Test)
  target_compile_definitions(MusicSorterAudioTests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME AudioTests COMMAND MusicSorterAudioTests -o "${CMAKE_BINARY_DIR}/audio-results.txt,txt" -o "-,txt")
  set_tests_properties(AudioTests PROPERTIES TIMEOUT 180)
endif()
install(TARGETS MusicSorter RUNTIME DESTINATION .)
qt_generate_deploy_app_script(TARGET MusicSorter OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
